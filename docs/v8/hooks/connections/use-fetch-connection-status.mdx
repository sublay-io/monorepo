## Overview

Returns a function that fetches the connection between the current authenticated user and a target user. The result says whether the two users are connected and, whenever a connection record exists between them (pending, accepted or declined), includes it.

`useConnectionManager` calls this automatically on mount. Use this hook directly when you need one-off status checks without managing state.

## Usage Example

```tsx
import { useFetchConnectionStatus } from "@sublay/react-js";

function ProfilePage({ userId }: { userId: string }) {
  const fetchConnectionStatus = useFetchConnectionStatus();

  useEffect(() => {
    fetchConnectionStatus({ userId }).then((result) => {
      if (result.isConnected) {
        console.log("Connected since", result.connection.respondedAt);
      } else if ("connection" in result) {
        // A pending or declined request; `type` says who sent it.
        console.log(result.connection.status, result.connection.type);
      } else {
        console.log("No connection");
      }
    });
  }, [userId]);
}
```

## Parameters

The hook returns a function. That function accepts:

<ParamField path="userId" type="string" required>
  The ID of the user to check. Cannot be the current user's own ID.
</ParamField>

## Returns

A `Promise<ConnectionStatus>` (see [ConnectionStatus](/v8/data-models/connection#connectionstatus)):

```ts
type ConnectionStatus =
  | { isConnected: false }
  | {
      isConnected: false;
      connection: ConnectionRow & { status: "pending" | "declined" };
    }
  | { isConnected: true; connection: ConnectionRow & { status: "accepted" } };
```

<ResponseField name="isConnected" type="boolean">
  `true` only when the connection is accepted.
</ResponseField>

<ResponseField name="connection" type="ConnectionRow | undefined">
  Present whenever a connection record exists between the two users, pending and declined included. `user` is the target user; `type` is `"sent"` when the current user sent the request and `"received"` when the target did; `status` is `"pending"`, `"accepted"` or `"declined"`. Absent when there is no connection record. See [ConnectionRow](/v8/data-models/connection#connectionrow).
</ResponseField>
