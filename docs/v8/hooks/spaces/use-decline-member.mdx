## Overview

`useDeclineMember` returns a callable function for declining a pending join request. Sets the membership status to `"rejected"`. Requires the caller to be a moderator or admin.

## Usage Example

```tsx
import { useDeclineMember } from "@sublay/react-js";

function PendingMemberRow({ spaceId, userId }: { spaceId: string; userId: string }) {
  const decline = useDeclineMember();

  return (
    <button onClick={() => decline({ spaceId, userId })}>
      Decline
    </button>
  );
}
```

## Parameters

<ParamField path="spaceId" type="string" required>
  UUID of the space.
</ParamField>

<ParamField path="userId" type="string" required>
  UUID of the user whose pending request to decline — their user id, not a membership id. Must be another member: your own id is refused (`400 space/own-member-id`). Your own membership is [`useCheckMyMembership`](/v8/hooks/spaces/use-check-my-membership) / [`useLeaveSpace`](/v8/hooks/spaces/use-leave-space).
</ParamField>

## Returns

<ResponseField name="message" type="string">
  Confirmation message.
</ResponseField>

<ResponseField name="membership" type="object">
  Updated membership with `id` and `status: "rejected"`.
</ResponseField>
