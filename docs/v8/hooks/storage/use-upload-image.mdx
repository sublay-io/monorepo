## Overview

Uploads an image to Sublay storage. The server processes the image, generates named variants according to the specified mode, and returns the [File](/v8/data-models/file) with proxy URLs for the original and every variant. The options are sent as one JSON field, `file.options`: the same options a record's own images take (for example `avatarFile` on [`useUser`](/v8/hooks/user/use-user)).

Works in both browser and React Native environments. Provides upload progress tracking.

The uploaded file isn't attached to anything. A file that belongs to a record is sent with that record's own hook (for example, the images of [`useCreateEntity`](/v8/hooks/entities/use-create-entity)).

<Note>Requires an authenticated user. The hook uses the private axios instance and will throw if no session is active.</Note>

## Usage Example

```tsx
import { useUploadImage } from "@sublay/react-js";

function AvatarUploader() {
  const { uploadImage, uploading, progress } = useUploadImage();

  const handleFile = async (file: File) => {
    const image = await uploadImage(file, {
      mode: "exact-dimensions",
      dimensions: {
        avatar: { width: 200, height: 200 },
        thumbnail: { width: 64, height: 64 },
      },
      format: "webp",
      quality: 85,
    });

    console.log(image.originalPath);
    console.log(image.image?.variants.avatar.publicPath);
  };

  return (
    <div>
      <input type="file" accept="image/*" onChange={e => e.target.files && handleFile(e.target.files[0])} />
      {uploading && <p>Uploading... {progress}%</p>}
    </div>
  );
}
```

## Parameters

`uploadImage(file, options)` accepts two arguments:

<ParamField path="file" type="BrowserFile | RNFile" required>
  The image to upload. In the browser, pass a standard `File` object. In React Native, pass `{ uri, name, type? }`.
</ParamField>

<ParamField path="options" type="UseUploadImageOptions" required>
  Processing configuration (`UploadImageOptions`) plus the call's own `onProgress` and `metadata`. Must include a `mode` field. See [Processing Modes](#processing-modes) below. Every requested size is 1–10000 px, and `dimensions`, `widths`, `heights` and `sizes` take at most 30 entries (one variant each).
</ParamField>

### Common Options (all modes)

<ParamField body="quality" type="number">
  JPEG/WebP quality, 1–100. Defaults to `85`.
</ParamField>

<ParamField body="format" type='"webp" | "jpeg" | "png" | "original"'>
  Output format for all variants. Defaults to `"webp"`. `original` keeps the upload's format when it is JPEG, PNG or WebP, and stores anything else (an AVIF or HEIC, for example) as WebP.
</ParamField>

<ParamField body="stripExif" type="boolean">
  Whether to remove EXIF metadata from the output. Defaults to `true`.
</ParamField>

<ParamField body="fit" type="string">
  How each variant fits its requested size. Defaults to `"inside"`. The values each mode takes are listed under [Processing Modes](#processing-modes).
</ParamField>

<ParamField body="originalMaxSize" type="number">
  The longest side, in px, the stored original is shrunk to (50–10000). Defaults to `3840`. An original already within it keeps its size. `originalSize`, `originalWidth` and `originalHeight` describe the stored original. A larger original takes more storage, which counts toward the project's file-storage limit.
</ParamField>

<ParamField body="pathParts" type="string[]">
  Folder segments for the stored image. The new file's id is always added after them (`<pathParts>/<fileId>/<filename>`, the file name taking the stored format's extension). Defaults to `["images"]`. A folder that is, or contains, `.` or `..` (also written `%2e%2e`), or that contains `?`, `#` or an encoded slash (`%2f`, `%5c`), is refused with a 400.
</ParamField>

<ParamField body="onProgress" type="(progress: number) => void">
  Called with upload percentage (0–100) during the multipart upload. Not sent to the server.
</ParamField>

<ParamField body="metadata" type="Record<string, any>">
  Custom key-value data stored on the file, sent as its own field (not inside `file.options`). The server adds `originalName` and `uploadedAt` alongside, replacing any values of yours under those keys.
</ParamField>

## Processing Modes

### `exact-dimensions`

Each variant has an explicit pixel width and height.

```ts
{
  mode: "exact-dimensions",
  dimensions: {
    thumbnail: { width: 64, height: 64 },
    card:      { width: 400, height: 300 },
  },
  fit?: "cover" | "contain" | "inside" | "outside", // default: "inside"
}
```

### `aspect-ratio-width-based`

All variants share a fixed aspect ratio. You specify widths; heights are computed.

```ts
{
  mode: "aspect-ratio-width-based",
  aspectRatio: { width: 16, height: 9 },
  widths: { small: 320, medium: 640, large: 1280 },
  fit?: "cover" | "contain" | "inside" | "outside",
}
```

### `aspect-ratio-height-based`

All variants share a fixed aspect ratio. You specify heights; widths are computed.

```ts
{
  mode: "aspect-ratio-height-based",
  aspectRatio: { width: 1, height: 1 },
  heights: { small: 100, medium: 200 },
  fit?: "cover" | "contain" | "inside" | "outside",
}
```

### `original-aspect`

Preserves the original image's aspect ratio. You specify the longest-edge size for each variant.

```ts
{
  mode: "original-aspect",
  sizes: { thumbnail: 100, medium: 400, full: 1200 },
  fit?: "inside" | "outside",
}
```

### `multi-aspect-ratio`

Generates variants across multiple aspect ratios with explicit sizes.

```ts
{
  mode: "multi-aspect-ratio",
  aspectRatios: [{ width: 1, height: 1 }, { width: 16, height: 9 }],
  sizes: { small: 200, large: 800 },
  fit?: "cover" | "contain" | "inside" | "outside",
}
```

## Returns

<ResponseField name="uploadImage" type="(file, options) => Promise<File>">
  Async function that performs the upload and resolves with the [File](/v8/data-models/file) on success.
</ResponseField>

<ResponseField name="uploading" type="boolean">
  `true` while the upload is in progress.
</ResponseField>

<ResponseField name="progress" type="number">
  Upload percentage (0–100). Resets to `0` when the upload finishes.
</ResponseField>

## Return Value Shape (`File`)

The [File](/v8/data-models/file), the same shape [Fetch File](/v8/api-reference/storage/fetch-file) returns, its `image` data included:

```ts
{
  id: string;
  userId: string | null;
  entityId: null; eventId: null; commentId: null; chatMessageId: null; spaceId: null;
  type: "image";
  originalPath: string;    // Proxy URL of the processed original
  originalSize: number;    // Bytes, of the stored original
  originalMimeType: string;
  position: number;
  metadata: Record<string, any>; // Yours, plus originalName and uploadedAt
  image: {
    originalWidth: number;   // Of the stored original
    originalHeight: number;
    variants: Record<string, FileImageVariant>; // Keyed by variant name
    processingStatus: "completed" | "failed";
    processingError: string | null;
    format: string;        // Output format
    quality: number;       // Output quality (85 when none was given)
    exifStripped: boolean;
    createdAt: string;
    updatedAt: string;
  } | null;
  createdAt: string;       // ISO 8601
  updatedAt: string;
}

// FileImageVariant
{
  publicPath: string; // Proxy URL
  width: number;
  height: number;
  size: number;       // Bytes
  format: string;
}
```
