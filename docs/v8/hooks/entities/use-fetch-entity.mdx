## Overview

`useFetchEntity` returns a function that fetches a single entity by its Sublay entity ID. Use this for on-demand fetches outside of `EntityProvider`.

## Usage Example

```tsx
import { useFetchEntity } from "@sublay/react-js";

function EntityLoader({ entityId }: { entityId: string }) {
  const fetchEntity = useFetchEntity();

  const load = async () => {
    const entity = await fetchEntity({ entityId, include: ["user", "topComments"], topCommentsLimit: 3 });
    console.log(entity);
  };

  return <button onClick={load}>Load Entity</button>;
}
```

## Parameters

<ParamField path="entityId" type="string" required>
  The Sublay entity ID to fetch.
</ParamField>

<ParamField path="include" type="string | EntityInclude[]">
  Optional. Populate related fields. Accepted values: `"user"`, `"space"`, `"topComments"`, `"saved"`, `"files"`, `"grants"`.
</ParamField>

<ParamField path="topCommentsLimit" type="number">
  With `"topComments"` included: how many top comments per entity. Server default `1`, maximum `5`. Sending it without `"topComments"` in `include` is rejected (`400`).
</ParamField>

<ParamField path="topCommentsSortByReaction" type="ReactionType">
  With `"topComments"` included: the reaction whose raw count ranks the top comments. Server default `"upvote"`. Sending it without `"topComments"` in `include` is rejected (`400`).
</ParamField>

## Returns

<ResponseField name="Entity" type="Entity">
  The fetched entity. When `spaceReputation` is requested, the embedded author carries an added `spaceReputation` number. See [Entity data model](/v8/data-models/entity) and [Reputation](/v8/data-models/reputation).
</ResponseField>
