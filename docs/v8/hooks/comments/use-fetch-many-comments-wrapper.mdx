## Overview

`useFetchManyCommentsWrapper` is a stateful wrapper around [`useFetchManyComments`](/v8/hooks/comments/use-fetch-many-comments) that manages the list, the cursor and the sort for you. It lists one level of a thread: an entity's top-level comments (`entityId`) or a comment's direct replies (`parentId`). Use it for a custom thread view outside `CommentSectionProvider`.

The hook doesn't fetch until `entityId` or `parentId` is set. Changing the sort, or any prop, resets the list and fetches from the start.

Nested replies (`replyDepth` above `0`) come back inside each comment's `replies` exactly as the server delivers them — the hook doesn't track them. For per-comment "load more replies", use `CommentSectionProvider` with [`useReplies`](/v8/hooks/comments/use-replies).

For a user's comments across entities, use [`useFetchCommentsByUserIdWrapper`](/v8/hooks/comments/use-fetch-comments-by-user-id-wrapper).

## Usage Example

```tsx
import { useFetchManyCommentsWrapper } from "@sublay/react-js";

function ReplyList({ commentId }: { commentId: string }) {
  const { comments, loading, error, hasMore, sortBy, setSortBy, loadMore } =
    useFetchManyCommentsWrapper({
      parentId: commentId,
      limit: 10,
      defaultSortBy: "createdAt",
      defaultSortDir: "asc",
      include: "user",
    });

  return (
    <div>
      <select value={sortBy} onChange={(e) => setSortBy(e.target.value as any)}>
        <option value="createdAt">Oldest</option>
        <option value="top">Top</option>
      </select>
      <ul>
        {comments.map((comment) => (
          <li key={comment.id}>{comment.content}</li>
        ))}
      </ul>
      {error && <p>Couldn't load replies.</p>}
      {hasMore && (
        <button onClick={() => loadMore()} disabled={loading}>
          Load more
        </button>
      )}
    </div>
  );
}
```

## Parameters

<ParamField body="entityId" type="string | null" optional>
  List this entity's top-level comments.
</ParamField>

<ParamField body="parentId" type="string | null" optional>
  List this comment's direct replies. With `parentId` set, `entityId` is ignored: the parent decides the entity.
</ParamField>

<ParamField body="limit" type="number" optional>
  Comments per page. Default: `10`.
</ParamField>

<ParamField body="defaultSortBy" type="CommentsSortByOptions" optional>
  Initial sort: `"createdAt"`, `"top"` or `"controversial"`. Default: `"createdAt"`.
</ParamField>

<ParamField body="defaultSortDir" type='"asc" | "desc"' optional>
  Initial sort direction for `"createdAt"` and `"top"`; ignored (not sent) for `"controversial"`, which ranks one way only. Default: `"desc"`.
</ParamField>

<ParamField body="defaultSortByReaction" type="ReactionType" optional>
  With `sortBy: "top"`: the reaction to rank by (raw count). Server default: `"upvote"`.
</ParamField>

<ParamField body="include" type='"user" | "grants" | ("user" | "grants")[]' optional>
  Related data to populate, at every level.
</ParamField>

<ParamField body="replyDepth" type="number" optional>
  Levels of replies to nest under each comment (`0`–`5`). Server default: `0`.
</ParamField>

<ParamField body="replyLimit" type="number" optional>
  Replies per comment in the nested levels. Server default: `3`.
</ParamField>

<ParamField body="replySortBy" type="CommentReplySortByOptions" optional>
  Sort for the nested levels. Defaults to the current sort.
</ParamField>

<ParamField body="replySortDir" type='"asc" | "desc"' optional>
  Sort direction for the nested levels. Defaults to the current direction; not sent when the nested sort is `"controversial"`.
</ParamField>

<ParamField body="replySortByReaction" type="ReactionType" optional>
  The reaction the nested levels rank by, when their sort is `"top"`.
</ParamField>

## Returns

<ResponseField name="comments" type="Comment[]">
  The fetched comments. Each `loadMore` appends the next page; a comment already in the list isn't added twice.
</ResponseField>

<ResponseField name="loading" type="boolean">
  `true` while a page is loading, and before `entityId` or `parentId` is first set. When both are removed after one was set, the list is cleared and `loading` is `false`.
</ResponseField>

<ResponseField name="error" type="Error | null">
  The last failure loading the list. Cleared when the next load starts. Call `loadMore()` to retry — `hasMore` stays `true` after a failure, and the retry asks for the same page again, including a first page that failed.
</ResponseField>

<ResponseField name="hasMore" type="boolean">
  `true` when more comments are available.
</ResponseField>

<ResponseField name="loadMore" type="() => Promise<void>">
  Loads the next page, or retries the page that failed. Does nothing while a page is loading or when `hasMore` is `false`, and never throws — failures land in `error`.
</ResponseField>

<ResponseField name="sortBy" type="CommentsSortByOptions">
  Current sort.
</ResponseField>

<ResponseField name="setSortBy" type="(sortBy: CommentsSortByOptions) => void">
  Changes the sort and reloads from the start.
</ResponseField>

<ResponseField name="sortDir" type='"asc" | "desc"'>
  Current sort direction.
</ResponseField>

<ResponseField name="setSortDir" type='(sortDir: "asc" | "desc") => void'>
  Changes the sort direction and reloads from the start.
</ResponseField>

<ResponseField name="sortByReaction" type="ReactionType | undefined">
  The reaction `"top"` ranks by. Only sent with `sortBy: "top"`.
</ResponseField>

<ResponseField name="setSortByReaction" type="(reaction: ReactionType | undefined) => void">
  Changes the ranking reaction. While `sortBy` is `"top"` this reloads from the start; under any other sort the value is kept for later and nothing reloads.
</ResponseField>
