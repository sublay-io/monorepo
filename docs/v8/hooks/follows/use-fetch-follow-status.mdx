## Overview

Returns a function that checks whether the current authenticated user follows a given `userId`. Resolves to a [`FollowStatus`](/v8/data-models/follow#followstatus): `isFollowing`, plus the follow row when they do.

## Usage Example

```tsx
import { useFetchFollowStatus } from "@sublay/react-js";

function ProfilePage({ userId }: { userId: string }) {
  const fetchFollowStatus = useFetchFollowStatus();

  useEffect(() => {
    fetchFollowStatus({ userId }).then((status) => {
      if (status.isFollowing) {
        console.log("Following since", status.follow.createdAt);
      }
    });
  }, [userId]);
}
```

## Parameters

The hook returns a function. That function accepts:

<ParamField path="userId" type="string" required>
  The ID of the user to check. Cannot be the current user's own ID.
</ParamField>

## Returns

The function resolves to a `FollowStatus`:

<ResponseField name="isFollowing" type="boolean">
  Whether the current user is following the target user.
</ResponseField>

<ResponseField name="follow" type="FollowRow | undefined">
  The follow row, present only when `isFollowing` is `true`. Its `user` is the target user. See [FollowRow](/v8/data-models/follow#followrow).
  <Expandable title="FollowRow properties">
    <ResponseField name="id" type="string">
      ID of the follow record. [useUnfollowByFollowId](/v8/hooks/follows/use-unfollow-by-follow-id) takes it.
    </ResponseField>
    <ResponseField name="user" type="User">
      The followed user. See [User data model](/v8/data-models/user).
    </ResponseField>
    <ResponseField name="createdAt" type="string">
      ISO timestamp of when the follow was created.
    </ResponseField>
  </Expandable>
</ResponseField>
