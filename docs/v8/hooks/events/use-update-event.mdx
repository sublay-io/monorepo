## Overview

`useUpdateEvent` returns a callable that edits an event. **Host-only** (`403 event/forbidden`): the logged-in user must be the owner or in `hostIds`, and, for an event in a space, an active member of that space who isn't banned in it or a space above it (`403 event/not-space-member`). The space's admins and moderators can't edit events they don't host. Everything to change goes under `update`; `hostIds`, `status`, and `spaceId` are not editable here. In `update`, pass `coverFile` to **replace** the cover, `images` and `files` to **append** to the event's files, and `removeFileIds` to remove the event's files (list files and/or the cover). Uploads send `multipart/form-data`; any file operation requires the `files-images` bundle.

## Usage Example

```tsx
import { useUpdateEvent } from "@sublay/react-js";

function EditEvent({ eventId }: { eventId: string }) {
  const updateEvent = useUpdateEvent();

  const handleSave = async (newCover: File) => {
    const event = await updateEvent({
      eventId,
      update: {
        capacity: 150,
        description: "Now with more space!",
        coverFile: { file: newCover, options: { mode: "original-aspect", sizes: { full: 1600 } } },
      },
    });
    console.log("Updated:", event.id);
  };

  const removeCover = async (coverImageId: string) => {
    await updateEvent({ eventId, update: { removeFileIds: [coverImageId] } });
  };
}
```

## Parameters

<ParamField path="eventId" type="string" required>The event to update.</ParamField>
<ParamField path="update" type="object" required>
  What to change. All fields are optional.
  <Expandable title="properties">
    <ParamField path="title, description, timezone, type, url, venueName, address, visibility, capacity, allowMaybe, guestListVisible, metadata">
      The event's mutable fields, as on [Update Event](/v8/api-reference/events/update-event).
    </ParamField>
    <ParamField path="startTime, endTime" type="string | Date">
      An ISO 8601 datetime in UTC, or a `Date` (sent as its ISO string).
    </ParamField>
    <ParamField path="location" type="{ latitude: number; longitude: number }">
      New coordinates.
    </ParamField>
    <ParamField path="coverFile" type="{ file: UploadFileInput; options: UploadImageOptions }">
      New cover image and its processing options (`mode` is required) — replaces the existing cover, whose file is deleted. `file` is a `File` or `Blob`, or a React Native file `{ uri, name, type? }`. Sends multipart.
    </ParamField>
    <ParamField path="images" type="{ files: UploadFileInput[]; options: UploadImageOptions }">
      Images to append after the event's existing files, processed into sizes per `options` (`mode` is required). A request takes at most 20 files in all. Sends multipart.
    </ParamField>
    <ParamField path="files" type="{ files: UploadFileInput[]; options?: { pathParts?: string[] } }">
      Files to append after the event's existing files, stored as uploaded. Sends multipart.
    </ParamField>
    <ParamField path="removeFileIds" type="string[]">
      Ids of the event's files to remove: list files and/or the cover (`coverImageId`; removing it clears the cover). Works alone or with uploads. Any other id refuses the whole update (`400 event/invalid-remove-file-ids`).
    </ParamField>
  </Expandable>
</ParamField>

## Returns

Returns a `Promise<Event>` — the updated [Event](/v8/data-models/event).
