## Overview

Returns a function that sends a connection request to another user. An optional message can be included with the request. The current user must be authenticated.

## Usage Example

```tsx
import { useRequestConnection } from "@sublay/react-js";

function ConnectButton({ userId }: { userId: string }) {
  const requestConnection = useRequestConnection();

  const handleConnect = async () => {
    const result = await requestConnection({ userId, message: "Let's connect!" });
    console.log("Request sent:", result.id);
  };

  return <button onClick={handleConnect}>Connect</button>;
}
```

## Parameters

The hook returns a function. That function accepts:

<ParamField path="userId" type="string" required>
  The ID of the user to send a connection request to.
</ParamField>

<ParamField path="message" type="string">
  An optional message to include with the connection request.
</ParamField>

## Returns

A `Promise<ConnectionRow>` that resolves to the new request, seen from the signed-in user's side (the requester). See [ConnectionRow](/v8/data-models/connection#connectionrow).

<ResponseField name="id" type="string">
  The connection's ID. Accept, decline and remove take it.
</ResponseField>

<ResponseField name="user" type="User">
  The user the request was sent to. Public profile fields only.
</ResponseField>

<ResponseField name="type" type='"sent" | "received"'>
  Always `"sent"`: the signed-in user sent the request.
</ResponseField>

<ResponseField name="message" type="string | null">
  The note sent with the request, or `null`.
</ResponseField>

<ResponseField name="status" type='"pending" | "accepted" | "declined"'>
  `"pending"` until the receiver responds.
</ResponseField>

<ResponseField name="createdAt" type="string">
  ISO timestamp of when the request was sent.
</ResponseField>

<ResponseField name="respondedAt" type="string | null">
  `null` while the request is pending.
</ResponseField>
