## Overview

`useFetchManyEntitiesWrapper` is a stateful wrapper around [`useFetchManyEntities`](/v8/hooks/entities/use-fetch-many-entities) that manages pagination, sorting, and list state for you. Use it for a self-contained entity feed kept in component state rather than in a shared, `listId`-keyed [`useEntityList`](/v8/hooks/entity-lists/use-entity-list) list — for example, a user profile feed, a filtered search results page, or any custom list with infinite scroll.

All filters and sort options mirror those of `useFetchManyEntities`. Changing any sort option automatically resets the list and re-fetches from page 1.

## Usage Example

```tsx
import { useFetchManyEntitiesWrapper } from "@sublay/react-js";

function UserFeed({ userId }: { userId: string }) {
  const {
    entities,
    loading,
    hasMore,
    sortBy,
    setSortBy,
    loadMore,
  } = useFetchManyEntitiesWrapper({
    userId,
    limit: 15,
    defaultSortBy: "createdAt",
    include: ["user"],
  });

  return (
    <div>
      <select value={sortBy} onChange={(e) => setSortBy(e.target.value as any)}>
        <option value="createdAt">Newest</option>
        <option value="top">Top</option>
        <option value="hot">Hot</option>
      </select>
      <ul>
        {entities.map((entity) => (
          <li key={entity.id}>{entity.title}</li>
        ))}
      </ul>
      {hasMore && (
        <button onClick={loadMore} disabled={loading}>
          Load more
        </button>
      )}
    </div>
  );
}
```

## Parameters

<ParamField body="limit" type="number">
  Number of entities per page: a whole number from `1` to `100`. Defaults to `10`. A larger value is rejected with `400 entity/invalid-query`, never lowered to `100`.
</ParamField>

<ParamField body="userId" type="string | null" optional>
  Filter to entities created by a specific user.
</ParamField>

<ParamField body="sourceId" type='string | "none" | null' optional>
  Only entities from this source, or `"none"` for entities with no source. Left out or `null`: any source.
</ParamField>

<ParamField body="spaceId" type='string | "none" | null' optional>
  Only entities in this space, or `"none"` for entities in no space. Left out or `null`: any space, or none.
</ParamField>

<ParamField body="followedOnly" type="boolean" optional>
  If `true`, return only entities from users the authenticated user follows.
</ParamField>

<ParamField body="include" type="EntityIncludeParam" optional>
  Populate related fields, as a single value or an array. Accepted values: `"user"`, `"space"`, `"topComments"`, `"saved"`, `"files"`, `"grants"`.
</ParamField>

<ParamField body="topCommentsLimit" type="number" optional>
  With `"topComments"` included: how many top comments per entity. Server default `1`, maximum `5`. Not sent when `include` doesn't contain `"topComments"`.
</ParamField>

<ParamField body="topCommentsSortByReaction" type="ReactionType" optional>
  With `"topComments"` included: the reaction whose raw count ranks the top comments. Server default `"upvote"`. Not sent when `include` doesn't contain `"topComments"`.
</ParamField>

<ParamField body="defaultSortBy" type="EntityListSortByOptions" optional>
  Initial sort field. Default: `"createdAt"`.
</ParamField>

<ParamField body="defaultSortByReaction" type="SortByReaction" optional>
  Initial reaction type `top` ranks by. Only sent while `sortBy` is `"top"`. Default: `"upvote"`.
</ParamField>

<ParamField body="defaultSortDir" type='"asc" | "desc"' optional>
  Initial sort direction. Default: `"desc"`; ignored (not sent) for `"hot"` and `"controversial"`, which rank one way only.
</ParamField>

<ParamField body="defaultSortType" type="SortType" optional>
  Initial sort algorithm. Options: `"auto"`, `"numeric"`, `"text"`, `"boolean"`, `"timestamp"`. Used when sorting by a metadata field. Default: `"auto"`.
</ParamField>

<ParamField body="timeFrame" type="TimeFrame | null" optional>
  Filter entities to a time window. Options: `"day"`, `"week"`, `"month"`, `"year"`.
</ParamField>

<ParamField body="keywordsFilters" type="KeywordsFilters | null" optional>
  Filter by keywords. See [Entity List Filters](/v8/sdk/entity-lists/filters).
</ParamField>

<ParamField body="titleFilters" type="TitleFilters | null" optional>
  Filter by title content. See [Entity List Filters](/v8/sdk/entity-lists/filters).
</ParamField>

<ParamField body="contentFilters" type="ContentFilters | null" optional>
  Filter by body content. See [Entity List Filters](/v8/sdk/entity-lists/filters).
</ParamField>

<ParamField body="attachmentsFilters" type="AttachmentsFilters | null" optional>
  Filter by attachments data. See [Entity List Filters](/v8/sdk/entity-lists/filters).
</ParamField>

<ParamField body="locationFilters" type="LocationFilters | null" optional>
  Filter by geographic proximity. See [Entity List Filters](/v8/sdk/entity-lists/filters).
</ParamField>

<ParamField body="metadataFilters" type="MetadataFilters | null" optional>
  Filter by metadata fields. See [Entity List Filters](/v8/sdk/entity-lists/filters).
</ParamField>

## Returns

<ResponseField name="entities" type="Entity[]">
  The current list of fetched entities. Appended to on each `loadMore` call.
</ResponseField>

<ResponseField name="loading" type="boolean">
  `true` while fetching the initial page or loading more.
</ResponseField>

<ResponseField name="error" type="Error | null">
  The last load failure, or `null`. Cleared when the next load starts; `loadMore()` retries it.
</ResponseField>

<ResponseField name="hasMore" type="boolean">
  `true` while `loadMore()` has something to fetch: a next page, or a load that failed.
</ResponseField>

<ResponseField name="sortBy" type="EntityListSortByOptions">
  Current sort field.
</ResponseField>

<ResponseField name="sortByReaction" type="SortByReaction">
  Current reaction type `top` ranks by. Only sent while `sortBy` is `"top"`.
</ResponseField>

<ResponseField name="sortDir" type='"asc" | "desc"'>
  Current sort direction.
</ResponseField>

<ResponseField name="sortType" type="SortType">
  Current sort algorithm.
</ResponseField>

<ResponseField name="setSortBy" type="(sortBy: EntityListSortByOptions) => void">
  Change the sort field. Resets the list and re-fetches from page 1.
</ResponseField>

<ResponseField name="setSortByReaction" type="(sortByReaction: SortByReaction) => void">
  Change the reaction type for reaction-based sorting. Resets and re-fetches.
</ResponseField>

<ResponseField name="setSortDir" type="(sortDir: 'asc' | 'desc') => void">
  Change the sort direction. Resets the list and re-fetches from page 1.
</ResponseField>

<ResponseField name="setSortType" type="(sortType: SortType) => void">
  Change the sort algorithm. Resets the list and re-fetches from page 1.
</ResponseField>

<ResponseField name="loadMore" type="() => Promise<void>">
  Load the next page of entities and append it, or retry the load that failed. Does nothing while a load is running or when there is nothing more. Never rejects: a failure lands in `error`.
</ResponseField>

<ResponseField name="refetch" type="() => Promise<void>">
  Reload from the first page with the current filters and sort. The loaded entities stay until the new first page replaces them. Never rejects: a failure lands in `error`.
</ResponseField>
